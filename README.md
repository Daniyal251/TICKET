Дистрибуция:
Какую модель монетизации выбираем?
Вариант 1: Единоразовая оплата за релиз (скриншот прилагается)
Сколько стоит дистрибуция сингла / EP / альбома? сингл 640 / EP 1290 / альбома 2090
Какой процент комиссии берем с доходов артистов? 0%, но порог вывода 5000 рублей, таким образом у 99% клиентов этот порог набираться не будет
Будет ли бесплатный тариф (ограниченный функционал)? нет, но нужна возможность ввода промокода для получения скидки и работы с рекламодателями.
Планируется ли подписочная модель с расширенным функционалом (как SFEROOM 
SEVEN)? Планируется раздел с пакетным родвижением, продажей треков под ключ и т.д. То есть доп функции, но не по подписке.
Продвижение:
6. Какие услуги продвижения будут доступны?
Питчинг в редакции площадок (Яндекс Музыка, VK, Spotify)? да
Пакеты плейлистов (гарантированное количество размещений)? да
Таргетированная реклама? да
Размещение у блогеров? да
Попадание в чарты? да
Будет ли продвижение оплачиваться отдельно или входить в подписку? ОТДЕЛЬНО. нужен спец раздел
Будет ли возможность автоматического продвижения (AI подбирает стратегию)? Непонятно, что это значит. В целом, звучит хайпово.
Финансы:
9. Как артисты будут получать выплаты? (карта, самозанятый, ИП, крипта?) Вопрос больше к Владимиру
10. Как часто осуществляются выплаты? (ежемесячно, ежеквартально, по запросу) Каждый месяц, но по факту деньги мало кому будут выводиться изза минимального порога вывода. Поэтому каждый месяц не будет сильно нагружать бухгалтерию
11. Есть ли минимальный порог для вывода средств? да, 5000 +-
12. Нужна ли функция сплит-роялти (автоматическое распределение между участниками)? да

- Загрузка нового релиза? да

- Профиль (настройки, документы)? да

    <div class="sh"><div class="sn">04</div><h2>База данных: DDL (PostgreSQL)</h2>

    <pre class="code"><span class="cm">-- ENUM'ы</span>
CREATE TYPE user_role      AS ENUM ('artist','manager','financier','superadmin');
CREATE TYPE release_type   AS ENUM ('single','maxi','ep','album');
CREATE TYPE release_status AS ENUM ('draft','moderation','rejected','sent','published','takedown');
CREATE TYPE comp_type      AS ENUM ('original','remix','cover');
CREATE TYPE part_kind      AS ENUM ('performer','feature','beatmaker','lyricist');
CREATE TYPE pay_status     AS ENUM ('pending','succeeded','failed','refunded');
CREATE TYPE promo_kind     AS ENUM ('pct','fix');
CREATE TYPE promo_scope    AS ENUM ('releases','services','all');
CREATE TYPE order_status   AS ENUM ('new','in_progress','done','rejected');
CREATE TYPE ticket_status  AS ENUM ('open','answered','closed');
CREATE TYPE payout_method  AS ENUM ('card','sbp','account');
CREATE TYPE payout_status  AS ENUM ('requested','approved','paid','declined');
CREATE TYPE mod_decision   AS ENUM ('approve','reject');

<span class="cm">-- Пользователи и договор</span>
CREATE TABLE users(
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  role user_role NOT NULL DEFAULT 'artist',
  email citext UNIQUE NOT NULL,
  password_hash text NOT NULL,
  artist_name varchar(60) NOT NULL,          <span class="cm">-- псевдоним: метаданные + мультиссылка</span>
  full_name text, phone text,
  email_confirmed_at timestamptz,
  tg_id bigint, status varchar(10) NOT NULL DEFAULT 'active',
  created_at timestamptz NOT NULL DEFAULT now(), updated_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE user_contracts(
  user_id uuid PRIMARY KEY REFERENCES users(id) ON DELETE CASCADE,
  birth_date date, legal_status varchar(10) NOT NULL DEFAULT 'phys', <span class="cm">-- phys|self_emp|iep</span>
  inn varchar(12), passport_enc bytea,       <span class="cm">-- шифрование at-rest</span>
  requisites_json jsonb,                     <span class="cm">-- карта/СБП/р/с</span>
  accepted_offer_at timestamptz);            <span class="cm">-- акцепт оферты + согласие 152-ФЗ</span>

<span class="cm">-- Справочники</span>
CREATE TABLE genres(id smallserial PRIMARY KEY, name text NOT NULL, sub jsonb NOT NULL DEFAULT '[]');
CREATE TABLE stores(id smallserial PRIMARY KEY, name text NOT NULL, active boolean NOT NULL DEFAULT true);

<span class="cm">-- Релизы и треки (поля п.1–10 анкеты + системные)</span>
CREATE TABLE releases(
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid NOT NULL REFERENCES users(id),
  type release_type NOT NULL,
  title text NOT NULL,                       <span class="cm">-- п.1 название трека (для single/maxi)</span>
  ep_title text,                             <span class="cm">-- п.2 название EP/альбома</span>
  genre_id smallint REFERENCES genres(id), subgenre text,
  explicit_bool boolean NOT NULL,            <span class="cm">-- п.7 мат да/нет → Explicit</span>
  comp comp_type NOT NULL DEFAULT 'original',<span class="cm">-- п.9</span>
  rights_url text, rights_confirmed boolean NOT NULL DEFAULT false,
  borrowings boolean NOT NULL DEFAULT false, borrowings_url text, <span class="cm">-- п.10</span>
  language text, lyrics text, pc_holder text,<span class="cm">-- рекомендуемые опции</span>
  release_date date, asap boolean NOT NULL DEFAULT true,
  status release_status NOT NULL DEFAULT 'draft',
  upc varchar(13), cover_key text,
  paid_payment_id uuid, reject_reason text, reject_comment text,
  created_at timestamptz NOT NULL DEFAULT now(), updated_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE tracks(
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  release_id uuid NOT NULL REFERENCES releases(id) ON DELETE CASCADE,
  position smallint NOT NULL,
  title text NOT NULL,                       <span class="cm">-- п.1 (список для EP/альбома)</span>
  audio_key text NOT NULL, duration_sec int, <span class="cm">-- п.6 хронометраж: авто из файла</span>
  isrc varchar(12) NOT NULL,                 <span class="cm">-- авто: RU-XXX-YY-NNNNN</span>
  audio_meta jsonb);                         <span class="cm">-- кодек/битрейт/битность/частота (пречек)</span>
CREATE TABLE participants(
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  release_id uuid NOT NULL REFERENCES releases(id) ON DELETE CASCADE,
  kind part_kind NOT NULL,                   <span class="cm">-- п.3 исполнитель/фит, п.4 битмейкер, п.5 автор текста</span>
  full_name text NOT NULL, alias text,
  share_pct smallint NOT NULL DEFAULT 0,     <span class="cm">-- сплит: сумма по релизу = 100</span>
  user_id uuid REFERENCES users(id));        <span class="cm">-- если участник зарегистрирован</span>
CREATE TABLE release_stores(
  release_id uuid REFERENCES releases(id) ON DELETE CASCADE,
  store_id smallint REFERENCES stores(id),
  delivery_status varchar(16) NOT NULL DEFAULT 'queued', external_id text, url text,

CREATE TABLE promocodes(

CREATE TABLE promo_redemptions(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

  payment_id uuid, created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE payments(
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid NOT NULL REFERENCES users(id),
  subject_type varchar(8) NOT NULL,          <span class="cm">-- release|order</span>
  subject_id uuid NOT NULL,
  amount_kop int NOT NULL, discount_kop int NOT NULL DEFAULT 0, promo_id uuid REFERENCES promocodes(id),
  provider varchar(16) NOT NULL DEFAULT 'yookassa', provider_id text,
  status pay_status NOT NULL DEFAULT 'pending', receipt_url text,
  created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE balances(user_id uuid PRIMARY KEY REFERENCES users(id), amount_kop bigint NOT NULL DEFAULT 0);
CREATE TABLE balance_ops(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES users(id), kind varchar(10) NOT NULL, <span class="cm">-- accrual|split|payout|refund</span>
  amount_kop bigint NOT NULL, ref jsonb, created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE payouts(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES users(id), amount_kop bigint NOT NULL,
  method payout_method NOT NULL, requisites_json jsonb NOT NULL,
  status payout_status NOT NULL DEFAULT 'requested', registry_month date,
  created_at timestamptz NOT NULL DEFAULT now());

<span class="cm">-- Модерация, мультиссылки, продвижение, тикеты, служба</span>
CREATE TABLE moderations(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  release_id uuid NOT NULL REFERENCES releases(id),
  manager_id uuid REFERENCES users(id),
  decision mod_decision, reason_code text, comment text,
  precheck_json jsonb,                       <span class="cm">-- результат авто-пречека</span>

CREATE TABLE multilinks(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

CREATE TABLE services(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL, description text, price_kop int NOT NULL,
  sla_days smallint, brief_schema jsonb NOT NULL DEFAULT '{}', active boolean NOT NULL DEFAULT true);
CREATE TABLE orders(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES users(id), release_id uuid REFERENCES releases(id),
  service_id uuid REFERENCES services(id), brief_json jsonb NOT NULL,
  payment_id uuid REFERENCES payments(id), status order_status NOT NULL DEFAULT 'new',
  manager_comment text, created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE pitchings(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  release_id uuid NOT NULL REFERENCES releases(id),
  story text NOT NULL, mood text, similar_artists text, targets_json jsonb,
  status varchar(16) NOT NULL DEFAULT 'new', created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE tickets(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id uuid REFERENCES users(id), subject text NOT NULL,
  status ticket_status NOT NULL DEFAULT 'open', created_at timestamptz NOT NULL DEFAULT now());
CREATE TABLE ticket_messages(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  ticket_id uuid REFERENCES tickets(id) ON DELETE CASCADE,
  author_id uuid NOT NULL, author_role user_role NOT NULL, body text NOT NULL,

CREATE TABLE royalty_reports(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),   <span class="cm">-- Фаза 2</span>

  release_id uuid REFERENCES releases(id), track_id uuid REFERENCES tracks(id),

  UNIQUE(period, store_id, release_id, track_id));
CREATE TABLE notifications_log(id uuid PRIMARY KEY DEFAULT gen_random_uuid(),

  payload jsonb, sent_at timestamptz NOT NULL DEFAULT now());
